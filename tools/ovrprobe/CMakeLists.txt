foreach(probe ovrprobe handprobe)
  if(probe STREQUAL "ovrprobe")
    add_executable(${probe} main.cpp)
  else()
    add_executable(${probe} ${probe}.cpp)
  endif()
  target_compile_features(${probe} PRIVATE cxx_std_20)
  target_compile_definitions(${probe} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(${probe} PRIVATE /W4 /permissive-)
  set_target_properties(${probe} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
